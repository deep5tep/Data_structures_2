using namespace std;
using Complex = complex<float>;

const int n = 4096;

// 1. Вариант: Наивный алгоритм (Линейная алгебра)
void multiply_naive(const vector<Complex>& A, const vector<Complex>& B, vector<Complex>& C) {
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            Complex sum(0, 0);
            for (int k = 0; k < n; ++k) {
                sum += A[i * n + k] * B[k * n + j];
            }
            C[i * n + j] = sum;
        }
    }
}

// 3. Вариант: Глубоко оптимизированный алгоритм (Блоки + SoA + AVX2 + FMA + Смена осей циклов + std::thread)
void multiply_optimized(const vector<Complex>& A, const vector<Complex>& B, vector<Complex>& C) {
    vector<float> A_real(n * n), A_imag(n * n);
    vector<float> B_real(n * n), B_imag(n * n);
    vector<float> C_real(n * n, 0.0f), C_imag(n * n, 0.0f);

    // Конвертируем AoS в SoA для линейного чтения в AVX регистрах
    for (int i = 0; i < n * n; ++i) {
        A_real[i] = A[i].real(); A_imag[i] = A[i].imag();
        B_real[i] = B[i].real(); B_imag[i] = B[i].imag();
    }

    const int BLOCK_SIZE = 64;
    unsigned int num_threads = thread::hardware_concurrency();
    if (num_threads == 0) num_threads = 4;

    vector<thread> threads;
    int rows_per_thread = n / num_threads;

    auto worker = [&](int start_i, int end_i) {
        // Итерируемся по блокам k (sk) наружу, чтобы максимизировать кэш-локальность для матрицы B
        for (int sk = 0; sk < n; sk += BLOCK_SIZE) {
            for (int sj = 0; sj < n; sj += BLOCK_SIZE) {

                for (int i = start_i; i < end_i; ++i) {
                    float* c_re_row = &C_real[i * n];
                    float* c_im_row = &C_imag[i * n];
                    const float* a_re_row = &A_real[i * n];
                    const float* a_im_row = &A_imag[i * n];

                    // Цикл по k внешний по отношению к j внутри подблока для утилизации регистров
                    for (int k = sk; k < sk + BLOCK_SIZE; ++k) {
                        __m256 a_re_vec = _mm256_set1_ps(a_re_row[k]);
                        __m256 a_im_vec = _mm256_set1_ps(a_im_row[k]);

                        const float* b_re_row = &B_real[k * n];
                        const float* b_im_row = &B_imag[k * n];

                        for (int j = sj; j < sj + BLOCK_SIZE; j += 8) {
                            __m256 c_re_vec = _mm256_loadu_ps(&c_re_row[j]);
                            __m256 c_im_vec = _mm256_loadu_ps(&c_im_row[j]);

                            __m256 b_re_vec = _mm256_loadu_ps(&b_re_row[j]);
                            __m256 b_im_vec = _mm256_loadu_ps(&b_im_row[j]);

                            // Комплексное умножение через аппаратный FMA
                            c_re_vec = _mm256_fmadd_ps(a_re_vec, b_re_vec, c_re_vec);
                            c_re_vec = _mm256_fnmadd_ps(a_im_vec, b_im_vec, c_re_vec);

                            c_im_vec = _mm256_fmadd_ps(a_re_vec, b_im_vec, c_im_vec);
                            c_im_vec = _mm256_fmadd_ps(a_im_vec, b_re_vec, c_im_vec);

                            _mm256_storeu_ps(&c_re_row[j], c_re_vec);
                            _mm256_storeu_ps(&c_im_row[j], c_im_vec);
                        }
                    }
                }

            }
        }
        };

    // Параллельное распределение строк по потокам
    for (unsigned int t = 0; t < num_threads; ++t) {
        int start_i = t * rows_per_thread;
        int end_i = (t == num_threads - 1) ? n : start_i + rows_per_thread;
        threads.push_back(thread(worker, start_i, end_i));
    }

    for (auto& th : threads) {
        if (th.joinable()) th.join();
    }

    // Собираем данные обратно в AoS
    for (int i = 0; i < n * n; ++i) {
        C[i] = Complex(C_real[i], C_imag[i]);
    }
}

// Вывод статистики в фиксированном формате чисел
double print_stats(string name, double seconds) {
    double c = 2.0 * n * n * n; // Оценка сложности по формуле лабы
    double p = (c / seconds) * 1e-6; // MFlops
    cout << name << ":\n";
    cout << "  Время: " << seconds << " s\n";
    cout << fixed << setprecision(2);
    cout << "  Производительность: " << p << " MFlops\n";
    cout << defaultfloat;
    return p;
}

// Математически верный вывод комплексных чисел в формате: a + bi
void print_matrix_sample(const vector<Complex>& M) {
    cout << "  Фрагмент матрицы (3x3):\n";
    cout << fixed << setprecision(2);

    for (int i = 0; i < 3; ++i) {
        cout << "    ";
        for (int j = 0; j < 3; ++j) {
            Complex val = M[i * n + j];
            float re = val.real();
            float im = val.imag();

            string cell = to_string(re).substr(0, to_string(re).find('.') + 3);
            if (im >= 0) {
                cell += " + " + to_string(im).substr(0, to_string(im).find('.') + 3) + "i";
            }
            else {
                cell += " - " + to_string(abs(im)).substr(0, to_string(abs(im)).find('.') + 3) + "i";
            }

            cout << setw(26) << left << cell << " ";
        }
        cout << "\n";
    }
    cout << "\n";
    cout << defaultfloat;
}
